**Create `AgentsUsageBar/UI/Welcome/DetectionProbe.swift`:**

Implement exactly as documented in Research Q8 and Q15. Key requirements:
- `DetectionResult` enum: `.detected`, `.notRunning`, `.notConfigured`, `.notDetected`
- `DetectionProbe.probeAll(config:localhostHTTP:fileManager:) async -> [ProviderID: DetectionResult]`
- `withTaskGroup` (non-throwing) for 7 concurrent tasks
- FS probes (OpenRouter, Claude, Codex, Gemini) are synchronous — wrapped in task
- HTTP probes (Ollama, LM Studio, llama.cpp) use `localhostHTTP` (2s timeout already configured)
- Reuse existing loaders (`ClaudeCredentialLoader`, `CodexCredentialLoader`, `GeminiCredentialLoader`, `GeminiSettingsGate`)
- NEVER read shell RC files (CFG-06 — no `~/.zshrc`, `~/.bashrc`, `config.fish` references)

```swift
import Foundation

/// Synchronous/async detection results for a single provider.
public enum DetectionResult: Sendable, Equatable {
    case detected       // Signal found; will be auto-enabled on first launch
    case notRunning     // HTTP probe reached host but service not running (localhost only)
    case notConfigured  // Required config (port, env key) is absent
    case notDetected    // FS probes found nothing (no credentials/config files)
}

/// Runs per-provider detection probes in parallel.
///
/// - FS-based probes: OpenRouter (env/TOML key), Claude, Codex, Gemini (credential files).
/// - HTTP-based probes: Ollama, LM Studio, llama.cpp (localhost endpoints within 2s).
/// - NEVER reads shell RC files (CFG-06 anti-feature).
public enum DetectionProbe {

    /// Runs all 7 provider detection probes concurrently.
    ///
    /// - Parameters:
    ///   - config: Current `AppConfig` (used for env/TOML values like apiKey, ports).
    ///   - localhostHTTP: The 2s-timeout `HTTPClient` instance from `AppDependencies`.
    ///     Pass the existing instance — do NOT create a new `URLSession`.
    ///   - fileManager: Injectable for test isolation. Defaults to `.default`.
    /// - Returns: Map of provider → detection result for all 7 providers.
    public static func probeAll(
        config: AppConfig,
        localhostHTTP: any HTTPClient,
        fileManager: FileManager = .default
    ) async -> [ProviderID: DetectionResult] {
        await withTaskGroup(of: (ProviderID, DetectionResult).self) { group in

            // 1. OpenRouter — env OR TOML api_key (synchronous; no HTTP needed for detection)
            group.addTask {
                let result: DetectionResult = config.openrouter.apiKey != nil ? .detected : .notConfigured
                return (.openrouter, result)
            }

            // 2. Claude — credential files or Keychain (reuse ClaudeCredentialLoader)
            group.addTask {
                // ClaudeCredentialLoader checks ~/.claude/.credentials.json + Keychain
                let result: DetectionResult = ClaudeCredentialLoader().loadCredentials() != nil
                    ? .detected
                    : probeClaudeFS(fileManager: fileManager)
                return (.claude, result)
            }

            // 3. Codex — ~/.codex/sessions/ exists OR ~/.codex/auth.json exists
            group.addTask {
                let result: DetectionResult = probeCodexFS(fileManager: fileManager)
                return (.codex, result)
            }

            // 4. Gemini — settings.json has oauth-personal AND oauth_creds.json present
            group.addTask {
                let result: DetectionResult = probeGeminiFS(fileManager: fileManager)
                return (.gemini, result)
            }

            // 5. Ollama — HTTP probe (2s timeout via localhostHTTP)
            group.addTask {
                let url = URL(string: "http://localhost:11434/api/version")!
                let result = await probeLocalHTTP(url: url, http: localhostHTTP)
                return (.ollama, result)
            }

            // 6. LM Studio — HTTP probe on configured port (default 1234)
            group.addTask {
                let url = URL(string: "http://localhost:\(config.lmstudio.port)/v1/models")!
                let result = await probeLocalHTTP(url: url, http: localhostHTTP)
                return (.lmstudio, result)
            }

            // 7. llama.cpp — requires configured port (LOCAL-03 no scanning)
            group.addTask {
                guard let port = config.llamacpp.port else {
                    return (.llamacpp, .notConfigured)
                }
                let url = URL(string: "http://localhost:\(port)/health")!
                let result = await probeLocalHTTP(url: url, http: localhostHTTP)
                return (.llamacpp, result)
            }

            var results: [ProviderID: DetectionResult] = [:]
            for await (id, result) in group {
                results[id] = result
            }
            return results
        }
    }

    // MARK: - FS probe helpers (synchronous)

    private static func probeClaudeFS(fileManager: FileManager) -> DetectionResult {
        let home = fileManager.homeDirectoryForCurrentUser.path
        let projectsPath = home + "/.claude/projects"
        let credPath = home + "/.claude/.credentials.json"
        if fileManager.fileExists(atPath: projectsPath) ||
           fileManager.fileExists(atPath: credPath) {
            return .detected
        }
        return .notDetected
    }

    private static func probeCodexFS(fileManager: FileManager) -> DetectionResult {
        let home = fileManager.homeDirectoryForCurrentUser.path
        let sessionsPath = home + "/.codex/sessions"
        let authPath = home + "/.codex/auth.json"
        if fileManager.fileExists(atPath: sessionsPath) ||
           fileManager.fileExists(atPath: authPath) {
            return .detected
        }
        return .notDetected
    }

    private static func probeGeminiFS(fileManager: FileManager) -> DetectionResult {
        let home = fileManager.homeDirectoryForCurrentUser.path
        let credsPath = home + "/.gemini/oauth_creds.json"
        let settingsPath = home + "/.gemini/settings.json"
        guard fileManager.fileExists(atPath: credsPath) else { return .notDetected }
        // GeminiSettingsGate.isOAuthPersonal checks nested security.auth.selectedType (STATE #70)
        guard GeminiSettingsGate.isOAuthPersonal(
            settingsPath: URL(fileURLWithPath: settingsPath),
            fileManager: fileManager
        ) else { return .notDetected }
        return .detected
    }

    // MARK: - HTTP probe helper (async)

    private static func probeLocalHTTP(url: URL, http: any HTTPClient) async -> DetectionResult {
        struct EmptyResponse: Decodable {}
        do {
            _ = try await http.get(
                URLRequest(url: url),
                as: EmptyResponse.self
            )
            return .detected
        } catch {
            // All localhost connection failures → .notRunning (not an error for detection purposes)
            return .notRunning
        }
    }
}
```

```swift
import Foundation

public enum OnboardingCopyError: Error, Sendable {
    case bundleResourceMissing
    case decodeFailed(Error)
}

/// Per-provider copy for the "How to enable" help panels.
/// Loaded from `Resources/Onboarding/providers.json` at runtime.
public struct ProviderOnboardingInfo: Decodable, Sendable {
    public let providerID: String
    public let displayName: String
    public let detectionDescription: String
    public let envSnippet: String
    public let envNote: String
    public let tomlSnippet: String
    public let tomlNote: String
    public let docsURL: String?
    // Lenient: unknown future fields silently ignored by JSONDecoder default behavior
}

public struct OnboardingCopy: Decodable, Sendable {
    public let providers: [ProviderOnboardingInfo]

    public static func loadBundled() throws -> OnboardingCopy {
        guard let url = Bundle.main.url(
            forResource: "providers",
            withExtension: "json",
            subdirectory: "Onboarding"
        ) else {
            throw OnboardingCopyError.bundleResourceMissing
        }
        do {
            let data = try Data(contentsOf: url)
            return try JSONDecoder().decode(OnboardingCopy.self, from: data)
        } catch {
            throw OnboardingCopyError.decodeFailed(error)
        }
    }
}
```

```swift
import SwiftUI
import AppKit

struct SettingsProvidersTab: View {

    @Environment(\.preferences) private var preferences
    @Environment(AggregateStore.self) private var store

    @State private var detectionResults: [ProviderID: DetectionResult] = [:]
    @State private var onboardingCopy: OnboardingCopy? = nil
    @State private var expandedRows: Set<ProviderID> = []
    @State private var isChecking: Bool = false

    var body: some View {
        ScrollView {
            VStack(alignment: .leading, spacing: 0) {
                ForEach(ProviderID.allKnown, id: \.rawValue) { providerID in
                    ProviderSettingsRow(
                        providerID: providerID,
                        isEnabled: preferences.providerEnabled[providerID] ?? true,
                        detectionResult: detectionResults[providerID],
                        onboardingInfo: onboardingCopy?.providers.first { $0.providerID == providerID.rawValue },
                        isExpanded: expandedRows.contains(providerID),
                        isChecking: isChecking,
                        onToggle: { enabled in
                            preferences.setProviderEnabled(providerID, enabled: enabled)
                        },
                        onRecheck: {
                            Task { await recheckProvider(providerID) }
                        },
                        onToggleExpand: {
                            if expandedRows.contains(providerID) {
                                expandedRows.remove(providerID)
                            } else {
                                expandedRows.insert(providerID)
                            }
                        }
                    )
                    Divider()
                }
            }
            .padding(.vertical, 8)
        }
        .frame(minWidth: 520, minHeight: 300)
        .onAppear {
            loadOnboardingCopy()
            Task { await runAllProbes() }
        }
    }

    // MARK: - Private

    private func loadOnboardingCopy() {
        guard onboardingCopy == nil else { return }
        onboardingCopy = try? OnboardingCopy.loadBundled()
    }

    private func runAllProbes() async {
        // Re-check all providers on tab appear (D-16).
        // Uses existing localhostHTTP from AppDependencies — but SettingsProvidersTab
        // only has access to @Environment stores. Create a fresh 2s HTTPClient here
        // for detection (acceptable: detection is once-per-tab-open, not polled).
        // Alternative: pass localhostHTTP via environment. Use simple approach for Phase 5.
        isChecking = true
        let config = ConfigStore(env: ProcessInfoEnvReader()).load()
        let http = URLSessionHTTPClient(timeoutSeconds: 2)
        detectionResults = await DetectionProbe.probeAll(config: config, localhostHTTP: http)
        isChecking = false
    }

    private func recheckProvider(_ id: ProviderID) async {
        let config = ConfigStore(env: ProcessInfoEnvReader()).load()
        let http = URLSessionHTTPClient(timeoutSeconds: 2)
        // Run all probes (withTaskGroup is already parallel; single provider re-probe
        // via full probeAll is acceptable for Phase 5 — 7 tasks, <=2s wall-clock).
        let results = await DetectionProbe.probeAll(config: config, localhostHTTP: http)
        detectionResults[id] = results[id]
    }
}

// MARK: - Per-provider row

private struct ProviderSettingsRow: View {

    let isEnabled: Bool
    let detectionResult: DetectionResult?
    let onboardingInfo: ProviderOnboardingInfo?
    let isExpanded: Bool
    let isChecking: Bool

    let onToggleExpand: () -> Void

    var body: some View {
        VStack(alignment: .leading, spacing: 0) {
            HStack {
                // Provider name + detection badge

                    Text(providerID.displayHint)
                        .fontWeight(.medium)

                    onRecheck()
                } label: {

                }
                .buttonStyle(.borderless)
                .disabled(isChecking)
                .accessibilityLabel("Re-check \(providerID.displayHint)")

                // Enable toggle

                    get: { isEnabled },
                    set: { onToggle($0) }
                ))
                .toggleStyle(.switch)
                .accessibilityLabel("Enable \(providerID.displayHint)")
            }
            .padding(.horizontal, 16)

            // Expandable "How to enable" panel (D-14) — only shown for undetected/unconfigured rows
            if onboardingInfo != nil && (detectionResult == .notDetected || detectionResult == .notConfigured || detectionResult == nil) {
                DisclosureGroup(

                    },
                    label: {
                        Text("How to enable")
                            .font(.caption)
                            .foregroundStyle(.secondary)
                            .padding(.horizontal, 16)
                    }
                )
            }
        }
    }

    @ViewBuilder
    private var detectionBadge: some View {
        if isChecking {
            Text("Checking…")
                .font(.caption)
                .foregroundStyle(.secondary)

            switch detectionResult {

                Label("Not detected", systemImage: "questionmark.circle")
                    .font(.caption)
                    .foregroundStyle(.secondary)
            case nil:
                EmptyView()
            }
        }
    }
```

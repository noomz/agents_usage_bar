**Create `AgentsUsageBar/App/WindowActivationObserver.swift`:**

Counter-based observer that flips `NSApplication.shared.setActivationPolicy` between `.regular` (while ≥1 managed window is open) and `.accessory` (when all managed windows are closed). Mirror `PowerObserver` exactly for Swift 6 compliance (`@MainActor public final class`, `nonisolated(unsafe)` tokens).

```swift
import AppKit
import os

/// Flips `NSApplication` activation policy between `.accessory` (no Dock icon) and
/// `.regular` (Dock icon visible) based on managed-window lifecycle events.
///
/// D-06 (CONTEXT.md): Settings window opens → `.regular`; closes → `.accessory`.
/// Counter-based so opening both Welcome + Settings simultaneously works correctly.
@MainActor
public final class WindowActivationObserver {

    // MARK: - State
    private var openWindowCount: Int = 0

    // nonisolated(unsafe): written once in init (@MainActor), read only in deinit (nonisolated).
    // NotificationCenter.removeObserver is thread-safe per Apple docs.
    nonisolated(unsafe) private let notificationCenter: NotificationCenter
    nonisolated(unsafe) private var openToken: NSObjectProtocol?
    nonisolated(unsafe) private var closeToken: NSObjectProtocol?

    private let logger = AppLogger.logger(category: "activation")

    // Window titles that trigger the flip. Matches SwiftUI Settings scene default title
    // and the WelcomeWindowController title set in Plan 05-05.
    // IMPORTANT: verify during execution that SwiftUI Settings window title == "Settings".
    // If SwiftUI uses a different title/identifier, adjust this set.
    private static let managedTitles: Set<String> = ["Settings", "Welcome"]

    // MARK: - Init

    public init(notificationCenter: NotificationCenter = NotificationCenter.default) {
        self.notificationCenter = notificationCenter

        openToken = notificationCenter.addObserver(
            forName: NSWindow.willOpenNotification,
            object: nil,
            queue: .main
        ) { [weak self] notification in
            Task { @MainActor [weak self] in
                self?.handleOpen(notification.object as? NSWindow)
            }
        }

        closeToken = notificationCenter.addObserver(
            forName: NSWindow.willCloseNotification,
            object: nil,
            queue: .main
        ) { [weak self] notification in
            Task { @MainActor [weak self] in
                self?.handleClose(notification.object as? NSWindow)
            }
        }
    }

    deinit {
        if let t = openToken  { notificationCenter.removeObserver(t) }
        if let t = closeToken { notificationCenter.removeObserver(t) }
    }

    // MARK: - Handlers

    private func isManaged(_ window: NSWindow?) -> Bool {
        guard let window else { return false }
        return Self.managedTitles.contains(window.title)
    }

    private func handleOpen(_ window: NSWindow?) {
        guard isManaged(window) else { return }
        openWindowCount += 1
        if openWindowCount == 1 {
            NSApplication.shared.setActivationPolicy(.regular)
            logger.notice("activation → .regular (openCount=\(self.openWindowCount))")
        }
    }

    private func handleClose(_ window: NSWindow?) {
        guard isManaged(window) else { return }
        openWindowCount = max(0, openWindowCount - 1)
        if openWindowCount == 0 {
            NSApplication.shared.setActivationPolicy(.accessory)
            logger.notice("activation → .accessory (openCount=0)")
        }
    }
}
```

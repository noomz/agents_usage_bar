// Observe UserPreferencesStore property changes and propagate to subsystems.
// Uses Swift @Observable's `withObservationTracking` pattern from AppKit context.
//
// Strategy: Use a long-lived Task that calls withObservationTracking to get
// called back whenever observed properties change. Since preferences is @Observable
// @MainActor, changes happen on the main actor and we hop via Task { @MainActor }.
//
// Implementation: Subscribe to UserDefaults.didChangeNotification (already handled
// inside UserPreferencesStore.loadAll). Wire explicit reactions in Dependencies via
// a periodic check or direct setter-side side-effects.
//
// Simplest correct approach for Phase 5: call the subsystems from UserPreferencesStore
// setters — but setters are in Config layer which shouldn't import Aggregation.
//
// Correct approach: Add a separate `observe(preferences:scheduler:store:)` helper
// that uses withObservationTracking in a loop to react to changes.

The cleanest Phase-5 approach: add a `startPreferencesObserver(preferences:scheduler:store:)` method to `AppDependencies` that is called from `AgentsUsageBarApp`'s `.task` modifier alongside `scheduler.start()`:

```swift
// In AppDependencies (static helper, not stored):
@MainActor
public static func observePreferences(
    _ preferences: UserPreferencesStore,
    scheduler: PollScheduler,
    store: AggregateStore
) async {
    // Use a loop with withObservationTracking to react to @Observable property changes.
    // This loop runs for the app's lifetime (cancelled when .task scope exits on scene teardown).
    var lastInterval = preferences.refreshInterval
    var lastThreshold = preferences.threshold

    while !Task.isCancelled {
        // withObservationTracking fires onChange once when any accessed property changes.
        await withObservationTracking {
            // Access the properties we want to observe:
            _ = preferences.refreshInterval
            _ = preferences.threshold
        } onChange: {
            // onChange fires on the thread that made the change (main actor here).
            // We just need the loop to re-evaluate — the actual work happens after.
        }

        if Task.isCancelled { break }

        // Re-read and react to changes:
        let newInterval = preferences.refreshInterval
        let newThreshold = preferences.threshold

        if newInterval != lastInterval {
            lastInterval = newInterval
            await scheduler.updateInterval(newInterval)
        }
        if newThreshold != lastThreshold {
            lastThreshold = newThreshold
            store.updateWarningFraction(newThreshold)
        }
    }
}
```

Note: `withObservationTracking(_:onChange:)` is the correct Swift `@Observable` observation API for non-SwiftUI contexts (macOS 14+, Observation framework). The `onChange` closure is called once when the tracked properties change; the outer loop re-subscribes immediately.

                            .foregroundStyle(.secondary)

                        value: Binding(

                            set: { preferences.setThreshold($0) }
                        ),
                        in: 0.5...0.95,
                        step: 0.05
                    )
                    .accessibilityLabel("Warning threshold")
                    .accessibilityValue("\(Int(preferences.threshold * 100)) percent")
                }
            }

            // MARK: Theme (D-07, Discretion)
            Section("Appearance") {
                Picker("Theme", selection: Binding(
                    get: { preferences.theme },
                    set: { preferences.setTheme($0) }
                )) {
                    Text("Light").tag(AppTheme.light)
                    Text("Dark").tag(AppTheme.dark)
                    Text("Auto").tag(AppTheme.auto)
                }
                .pickerStyle(.segmented)
            }

            // MARK: Open at Login (D-07, D-08, CFG-05)
            Section("System") {
                VStack(alignment: .leading, spacing: 4) {
                    Toggle("Open at login", isOn: Binding(
                        get: { preferences.openAtLogin },
                        set: { newValue in
                            Task { @MainActor in
                                await toggleOpenAtLogin(newValue)
                            }
                        }
                    ))

                    // Show status subtitle for .requiresApproval and .notFound
                    switch loginItemStatus {
                    case .requiresApproval:
                        HStack(spacing: 4) {
                            Text("Approve in System Settings")
                                .font(.caption)
                                .foregroundStyle(.secondary)
                            Button("Open Login Items ↗") {
                                openLoginItemsSettings()
                            }
                            .font(.caption)
                            .buttonStyle(.link)
                        }
                    case .notFound:
                        Text("Configuration error — see app bundle")
                            .font(.caption)
                            .foregroundStyle(.red)
                    case .enabled:
                        Text("Opens at login")
                            .font(.caption)
                            .foregroundStyle(.secondary)
                    default:
                        EmptyView()
                    }
                }
            }
        }
        .formStyle(.grouped)
        .padding()
        .onAppear {
            loginItemStatus = SMAppService.mainApp.status
        }
    }

    // MARK: - Actions

    @MainActor
    private func toggleOpenAtLogin(_ newValue: Bool) async {
        do {
            if newValue {
                try SMAppService.mainApp.register()
            } else {
                try SMAppService.mainApp.unregister()
            }
            // Only persist the new value when the system call succeeds
            preferences.setOpenAtLogin(newValue)

            // Silent revert — macOS norm (CONTEXT Discretion). UI toggle reverts because
            // preferences.openAtLogin still holds old value (setOpenAtLogin was not called).
            logger.warning("SMAppService toggle failed: \(error.localizedDescription, privacy: .public)")
        }
        // Re-read authoritative status (may move to .requiresApproval on first call)
        loginItemStatus = SMAppService.mainApp.status
    }

    private func openLoginItemsSettings() {
        // D-08: deep-link to System Settings → Login Items pane
        if let url = URL(string: "x-apple.systempreferences:com.apple.LoginItems-Settings.extension") {
            NSWorkspace.shared.open(url)
        }
    }
}

/// - When disabled: cancels any in-flight fetch Task for the provider and
///   removes it from the visible row set so the popover hides the row.
/// - When enabled: re-includes the provider on the next scheduler tick.

/// changes. SettingsProvidersTab only mutates the UserDefaults flag via

public func setProviderEnabled(_ providerID: ProviderID, enabled: Bool) {

        // Cancel any in-flight Task for this provider via structured concurrency.
        // (Exact cancellation mechanism depends on existing AggregateStore Task bookkeeping —
        //  executor must verify the existing Task-per-provider map and call cancel() on it.)
        cancelInFlightTask(for: providerID)
        // Remove from visible rows so popover hides the row immediately.
        markProviderDisabled(providerID)

        // Re-include: clear the disabled flag; next scheduler tick will fetch this provider.
        markProviderEnabled(providerID)
    }
}
